        // Customization Modal Logic
        function openCustomizeModal(productId) {
            const product = menuProducts.find(p => p.id === productId);
            if (!product) return;

            activeCustomizingProduct = product;
            document.getElementById('modalImg').src = product.image;
            document.getElementById('modalTitle').innerText = product.title;
            document.getElementById('modalCategory').innerText = product.category.toUpperCase();
            document.getElementById('modalBasePrice').innerText = `$${product.price.toFixed(2)}`;

            // Toggle sections based on type
            const isBeans = product.category === 'beans';
            const isBakery = product.category === 'bakery';

            document.getElementById('sizeSection').style.display = isBeans ? 'none' : 'block';
            document.getElementById('milkSection').style.display = (isBeans || isBakery) ? 'none' : 'block';
            document.getElementById('sweetnessSection').style.display = (isBeans || isBakery) ? 'none' : 'block';

            updateModalPrice();
            document.getElementById('customizeModal').classList.remove('hidden');
        }

        function closeCustomizeModal() {
            document.getElementById('customizeModal').classList.add('hidden');
        }

        function updateModalPrice() {
            if (!activeCustomizingProduct) return;
            let price = activeCustomizingProduct.price;

            const selectedSize = document.querySelector('input[name="size"]:checked');
            if (selectedSize && activeCustomizingProduct.category !== 'beans') {
                if (selectedSize.value.includes('Mediano')) price += 0.75;
                if (selectedSize.value.includes('Grande')) price += 1.25;
            }

            document.getElementById('modalCalculatedTotal').innerText = `$${price.toFixed(2)}`;
        }

        function confirmAddToCart(event) {
            event.preventDefault();
            if (!activeCustomizingProduct) return;

            let finalPrice = activeCustomizingProduct.price;
            const selectedSize = document.querySelector('input[name="size"]:checked');
            if (selectedSize && activeCustomizingProduct.category !== 'beans') {
                if (selectedSize.value.includes('Mediano')) finalPrice += 0.75;
                if (selectedSize.value.includes('Grande')) finalPrice += 1.25;
            }

            const isBeverage = activeCustomizingProduct.category !== 'beans' && activeCustomizingProduct.category !== 'bakery';

            const cartItem = {
                id: Date.now(),
                title: activeCustomizingProduct.title,
                price: finalPrice,
                size: isBeverage ? selectedSize.value : 'Estándar',
                milk: isBeverage ? document.getElementById('milkSelect').value : 'N/A',
                sweetness: isBeverage ? document.getElementById('sweetnessSelect').value : 'N/A',
                notes: document.getElementById('specialNotes').value,
                quantity: 1,
                image: activeCustomizingProduct.image
            };

            cart.push(cartItem);
            updateCart();
            closeCustomizeModal();
            showToast(`¡${cartItem.title} agregado al pedido!`);
        }

        function updateCartBadge() {
            const badge = document.getElementById('cartBadge');
            const totalQty = cart.reduce((sum, item) => sum + item.quantity, 0);
            badge.innerText = totalQty;
            if (totalQty > 0) {
                badge.classList.remove('scale-0');
                badge.classList.add('scale-100');
            } else {
                badge.classList.remove('scale-100');
                badge.classList.add('scale-0');
            }
        }

        function renderCartItems() {
            const container = document.getElementById('cartItemsContainer');
            const checkoutBtn = document.getElementById('checkoutBtn');

            if (cart.length === 0) {
                container.innerHTML = `
                    <div class="text-center py-16 text-aura-mocha/60">
                        <i class="fa-solid fa-mug-hot text-4xl mb-3 text-aura-pinkRose"></i>
                        <p class="text-sm">Tu pedido está vacío.</p>
                        <p class="text-xs mt-1 text-aura-mocha/40">Explora nuestro menú y añade tu bebida favorita.</p>
                    </div>
                `;
                checkoutBtn.disabled = true;
                calculateCartTotals(0);
                return;
            }

            checkoutBtn.disabled = false;
            let subtotal = 0;

            container.innerHTML = cart.map((item, index) => {
                subtotal += item.price * item.quantity;
                return `
                    <div class="bg-white p-4 rounded-2xl border border-aura-pinkMid/20 flex gap-3 items-center">
                        <img src="${item.image}" class="w-14 h-14 rounded-xl object-cover bg-aura-beige shrink-0" onerror="this.src='https://placehold.co/100x100/E8B4B8/2D1E18?text=Aura'">
                        <div class="flex-1 min-w-0">
                            <h4 class="font-serif font-bold text-xs text-aura-espresso truncate">${item.title}</h4>
                            <p class="text-[10px] text-aura-mocha/60 truncate">${item.size} • ${item.milk}</p>
                            <span class="text-xs font-semibold text-aura-pinkRose">$${(item.price * item.quantity).toFixed(2)}</span>
                        </div>
                        
                        <div class="flex items-center gap-2">
                            <button onclick="changeQty(${index}, -1)" class="w-6 h-6 rounded-full bg-aura-pinkSoft text-aura-espresso text-xs font-bold flex items-center justify-center hover:bg-aura-pinkMid">-</button>
                            <span class="text-xs font-bold text-aura-espresso w-4 text-center">${item.quantity}</span>
                            <button onclick="changeQty(${index}, 1)" class="w-6 h-6 rounded-full bg-aura-pinkSoft text-aura-espresso text-xs font-bold flex items-center justify-center hover:bg-aura-pinkMid">+</button>
                        </div>
                    </div>
                `;
            }).join('');

            calculateCartTotals(subtotal);
        }

        function changeQty(index, delta) {
            cart[index].quantity += delta;
            if (cart[index].quantity <= 0) {
                cart.splice(index, 1);
            }
            updateCart();
        }

            const val = document.getElementById('promoInput').value.trim().toUpperCase();

            const discountAmount = subtotal * appliedDiscount;

            document.getElementById('cartDiscount').innerText = `-$${discountAmount.toFixed(2)}`;
            document.getElementById('cartTotal').innerText = `$${total.toFixed(2)}`;
            document.getElementById('checkoutModalTotal').innerText = `$${total.toFixed(2)}`;
        }

                <h3 class="font-serif text-xl sm:text-2xl font-bold text-aura-espresso text-center mb-6">${q.question}</h3>

        function answerQuiz(drinkId) {
            quizAnswers.push(drinkId);
            if (currentQuizIndex < quizQuestions.length - 1) {
                currentQuizIndex++;
                renderQuizStep();
            } else {
                showQuizResult();
            }
        }

        // Ambient Audio Synthesizer (Coffee Shop Lounge Hum using Web Audio API)
        function toggleAmbientAudio() {
            const btn = document.getElementById('ambientToggleBtn');
            const icon = document.getElementById('soundIcon');
            const label = document.getElementById('soundLabel');
            const eq = document.getElementById('eqBars');

            if (!isAudioPlaying) {
                try {
                    const AudioContext = window.AudioContext || window.webkitAudioContext;
                    audioContext = new AudioContext();
                    
                    // Create soft ambient drone/lounge pad sound
                    const osc = audioContext.createOscillator();
                    const gain = audioContext.createGain();
                    
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(220, audioContext.currentTime); // Soft A3 note
                    
                    gain.gain.setValueAtTime(0.02, audioContext.currentTime); // Low non-intrusive volume
                    
                    osc.connect(gain);
                    gain.connect(audioContext.destination);
                    
                    osc.start();
                    ambientOscillator = osc;
                    isAudioPlaying = true;

                    icon.className = "fa-solid fa-volume-high text-aura-pinkRose";
                    label.innerText = "Aura Ambiance (On)";
                    eq.classList.remove('hidden');
                    eq.classList.add('flex');
                    showToast('♪ Reproduciendo ambiente relajante Aura');
                } catch(e) {
                    showToast('Audio no soportado en este navegador.');
                }
            } else {
                if (ambientOscillator) {
                    ambientOscillator.stop();
                    ambientOscillator.disconnect();
                }
                isAudioPlaying = false;
                icon.className = "fa-solid fa-volume-xmark text-aura-pinkRose";
                label.innerText = "Aura Lounge Audio";
                eq.classList.add('hidden');
                eq.classList.remove('flex');
            }
        }
